# BMC-K4510 on a Raspberry Pi 3B+ (bare metal, Circle + circle-libsdl2)
# The TV is driven at 1280x720; the presentation core scales the 640x480
# picture to 960x720 (4:3 kept, no distortion). No firmware margins: if the
# TV crops the edges, set its picture mode to "Just Scan" / "Dot by Dot".
# hdmi_mode=1 (640x480) also works and is lighter on the presentation core.
hdmi_group=1
hdmi_mode=4
disable_overscan=1

arm_64bit=1
kernel_address=0x80000
initial_turbo=0

[pi3]
kernel=kernel8.img

[pi3+]
kernel=kernel8.img
